FORM SEND_REQUEST_PIPO_DATA  USING W_BEGIN.

  IF S_UP_DT IS NOT INITIAL.
    W_CAL_DATE = S_UP_DT-LOW.
    CONCATENATE S_UP_DT-LOW+6(2) '-'
                S_UP_DT-LOW+4(2) '-'
                S_UP_DT-LOW+0(4) INTO W_FROMUPLOADEDDATE.

    IF S_UP_DT-HIGH IS NOT INITIAL.
      CONCATENATE S_UP_DT-HIGH+6(2) '-'
                  S_UP_DT-HIGH+4(2) '-'
                  S_UP_DT-HIGH+0(4) INTO W_TOUPLOADEDDATE.
    ELSE.
      CONCATENATE S_UP_DT-LOW+6(2) '-'
                  S_UP_DT-LOW+4(2) '-'
                  S_UP_DT-LOW+0(4) INTO W_TOUPLOADEDDATE.
    ENDIF.
  ELSE.

    CALL FUNCTION 'RP_CALC_DATE_IN_INTERVAL'
      EXPORTING
        DATE      = SY-DATUM
        DAYS      = '01'
        MONTHS    = W_MONTHS
        SIGNUM    = '-'
        YEARS     = W_YEARS
      IMPORTING
        CALC_DATE = W_CAL_DATE.

    CONCATENATE W_CAL_DATE+6(2) '-'
                W_CAL_DATE+4(2) '-'
                W_CAL_DATE+0(4) INTO W_FROMUPLOADEDDATE.

    CONCATENATE W_CAL_DATE+6(2) '-'
          W_CAL_DATE+4(2) '-'
          W_CAL_DATE+0(4) INTO W_TOUPLOADEDDATE.
  ENDIF.

  IF S_MONTH IS NOT INITIAL.
    IF S_MONTH-HIGH IS INITIAL.
      S_MONTH-HIGH = S_MONTH-LOW.
    ENDIF.
  ENDIF.

  CLEAR : IT_LOCATIONS.

  IT_OUTPUT_AUTH2-MT_GSTR2A_RECONCILIATION_REQ-AUTH_TOKEN     = W_TOKEN.
  IT_OUTPUT_AUTH2-MT_GSTR2A_RECONCILIATION_REQ-SOURCE         = C_CYG_2A.

  IT_OUTPUT_AUTH2-MT_GSTR2A_RECONCILIATION_REQ-SIZE                      = P_2.
  IT_OUTPUT_AUTH2-MT_GSTR2A_RECONCILIATION_REQ-START                     = W_BEGIN.
  CONDENSE IT_OUTPUT_AUTH2-MT_GSTR2A_RECONCILIATION_REQ-START.
  IT_OUTPUT_AUTH2-MT_GSTR2A_RECONCILIATION_REQ-FROM_RECONCILIATION_DATE  = W_FROMUPLOADEDDATE.
  IT_OUTPUT_AUTH2-MT_GSTR2A_RECONCILIATION_REQ-TO_RECONCILIATION_DATE    = W_TOUPLOADEDDATE.
  IT_OUTPUT_AUTH2-MT_GSTR2A_RECONCILIATION_REQ-IS_AGGREGATED_ITEMS    = 'true'.

        EXPORTING
          OUTPUT = IT_OUTPUT_AUTH2
        IMPORTING
          INPUT  = IT_INPUT_AUTH2.
